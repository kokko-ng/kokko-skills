# file-operations.txt - File and directory deletion patterns

# rm commands - file/directory deletion
(^|[[:space:]])rm[[:space:]]+-rf
(^|[[:space:]])rm[[:space:]]+-fr
(^|[[:space:]])rm[[:space:]]+-r[[:space:]]+-f
(^|[[:space:]])rm[[:space:]]+-f[[:space:]]+-r
(^|[[:space:]])rm[[:space:]]+--recursive[[:space:]]+--force
(^|[[:space:]])rm[[:space:]]+--force[[:space:]]+--recursive
(^|[[:space:]])rm[[:space:]]+-r[[:space:]]
(^|[[:space:]])rm[[:space:]]+--recursive
(^|[[:space:]])rm[[:space:]]+-f[[:space:]]
(^|[[:space:]])rm[[:space:]]+--force
\brm[[:space:]]+/
\brm[[:space:]]+\*
\brm[[:space:]]+\.\.
\brm[[:space:]]+~
\brm[[:space:]]+\$HOME

# Directory removal
(^|[[:space:]])rmdir[[:space:]]+

# Secure deletion utilities
(^|[[:space:]])shred[[:space:]]+
(^|[[:space:]])srm[[:space:]]+
(^|[[:space:]])wipe[[:space:]]+
(^|[[:space:]])secure-delete[[:space:]]+

# Truncate files (redirects are only flagged when they target system paths;
# a redirect into /tmp or a project directory is routine)
(^|[[:space:]])truncate[[:space:]]+-s[[:space:]]*0
(^|[[:space:]])truncate[[:space:]]+--size[[:space:]]*0
(^|[[:space:]])>[[:space:]]*/(etc|boot|dev/sd|usr|bin|sbin|lib)
(^|[[:space:]])cat[[:space:]]+/dev/null[[:space:]]*>

# Move to /dev/null (data loss)
(^|[[:space:]])mv[[:space:]]+.*[[:space:]]+/dev/null

# Overwrite files
(^|[[:space:]])cp[[:space:]]+/dev/null[[:space:]]+
(^|[[:space:]])dd[[:space:]]+of=.*[[:space:]]+if=/dev/zero
(^|[[:space:]])dd[[:space:]]+of=.*[[:space:]]+if=/dev/urandom

# Archive with source removal
(^|[[:space:]])zip[[:space:]]+-m
(^|[[:space:]])zip[[:space:]]+--move
(^|[[:space:]])tar[[:space:]]+--remove-files
(^|[[:space:]])7z[[:space:]]+d[[:space:]]

# Dangerous find -delete
(^|[[:space:]])find[[:space:]]+.*-delete
(^|[[:space:]])find[[:space:]]+.*-exec[[:space:]]+rm

# Core system files (extremely dangerous)
\brm[[:space:]]+.*\.bashrc
\brm[[:space:]]+.*\.zshrc
\brm[[:space:]]+.*\.profile
\brm[[:space:]]+.*\.bash_profile
\brm[[:space:]]+.*\.gitconfig
\brm[[:space:]]+.*\.ssh
\brm[[:space:]]+.*\.gnupg
\brm[[:space:]]+.*\.aws
\brm[[:space:]]+.*\.kube
\brm[[:space:]]+.*\.config

# Clearing temp and cache (can break running processes)
(^|[[:space:]])rm[[:space:]]+-rf[[:space:]]+/tmp
(^|[[:space:]])rm[[:space:]]+-rf[[:space:]]+/var/tmp
(^|[[:space:]])rm[[:space:]]+-rf[[:space:]]+.*\.cache
