# process.txt - Process control and termination operations

# Kill commands
(^|[[:space:]])kill[[:space:]]+-9
(^|[[:space:]])kill[[:space:]]+-SIGKILL
(^|[[:space:]])kill[[:space:]]+-KILL
(^|[[:space:]])kill[[:space:]]+-9[[:space:]]+1\b
(^|[[:space:]])kill[[:space:]]+-9[[:space:]]+-1

# Kill all processes by name
(^|[[:space:]])killall[[:space:]]+
(^|[[:space:]])pkill[[:space:]]+
(^|[[:space:]])pkill[[:space:]]+-9
(^|[[:space:]])pkill[[:space:]]+-SIGKILL

# Kill process groups
(^|[[:space:]])kill[[:space:]]+-[0-9]+[[:space:]]+-
(^|[[:space:]])killpg[[:space:]]+

# Force kill user processes
(^|[[:space:]])fuser[[:space:]]+-k
(^|[[:space:]])fuser[[:space:]]+-KILL
(^|[[:space:]])fuser[[:space:]]+-9

# X11 process killing
(^|[[:space:]])xkill

# Terminate by PID file
(^|[[:space:]])start-stop-daemon[[:space:]]+--stop

# Process control
(^|[[:space:]])renice[[:space:]]+.*[[:space:]]+-p[[:space:]]+1
(^|[[:space:]])ionice[[:space:]]+.*-p[[:space:]]+1

# Memory management
(^|[[:space:]])sync[[:space:]]+&&[[:space:]]+echo[[:space:]]+3[[:space:]]+>[[:space:]]*/proc/sys/vm/drop_caches
(^|[[:space:]])echo[[:space:]]+[123][[:space:]]+>[[:space:]]*/proc/sys/vm/drop_caches

# OOM killer manipulation
(^|[[:space:]])echo[[:space:]]+-17[[:space:]]+>[[:space:]]*/proc/.*/oom_adj
(^|[[:space:]])echo[[:space:]]+-1000[[:space:]]+>[[:space:]]*/proc/.*/oom_score_adj

# Control groups (cgroups)
(^|[[:space:]])cgdelete[[:space:]]+
(^|[[:space:]])systemd-cgtop[[:space:]]+--kill

# Process namespace destruction
(^|[[:space:]])unshare[[:space:]]+--kill-child
(^|[[:space:]])nsenter[[:space:]]+.*kill

# Container process killing
(^|[[:space:]])docker[[:space:]]+exec[[:space:]]+.*kill
(^|[[:space:]])kubectl[[:space:]]+exec[[:space:]]+.*kill

# Debugger that can kill
(^|[[:space:]])gdb[[:space:]]+.*-ex[[:space:]]+"kill"
(^|[[:space:]])lldb[[:space:]]+.*-o[[:space:]]+"kill"

# Trace/debug interruption
(^|[[:space:]])strace[[:space:]]+-p[[:space:]]+.*-k
(^|[[:space:]])ltrace[[:space:]]+-p[[:space:]]+.*-k

# Job control
(^|[[:space:]])disown[[:space:]]+-a
(^|[[:space:]])jobs[[:space:]]+-p[[:space:]]*\|[[:space:]]*xargs[[:space:]]+kill
