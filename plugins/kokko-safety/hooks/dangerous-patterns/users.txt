# users.txt - User and group management operations

# User deletion
(^|[[:space:]])userdel[[:space:]]+
(^|[[:space:]])deluser[[:space:]]+
(^|[[:space:]])user[[:space:]]+del[[:space:]]+

# User modification (can lock out)
(^|[[:space:]])usermod[[:space:]]+-L
(^|[[:space:]])usermod[[:space:]]+--lock
(^|[[:space:]])usermod[[:space:]]+-e[[:space:]]+1
(^|[[:space:]])passwd[[:space:]]+-l
(^|[[:space:]])passwd[[:space:]]+--lock
(^|[[:space:]])passwd[[:space:]]+-d
(^|[[:space:]])passwd[[:space:]]+--delete

# Group deletion
(^|[[:space:]])groupdel[[:space:]]+
(^|[[:space:]])delgroup[[:space:]]+

# Remove user from groups
(^|[[:space:]])gpasswd[[:space:]]+-d
(^|[[:space:]])usermod[[:space:]]+-G[[:space:]]+""

# Modify /etc/passwd or /etc/shadow directly
(^|[[:space:]])vipw
(^|[[:space:]])vigr
(^|[[:space:]])chpasswd
(^|[[:space:]])newusers

# Direct edits to auth files (extremely dangerous)
(^|[[:space:]])echo[[:space:]]+.*>>[[:space:]]*/etc/passwd
(^|[[:space:]])echo[[:space:]]+.*>>[[:space:]]*/etc/shadow
(^|[[:space:]])echo[[:space:]]+.*>>[[:space:]]*/etc/group
(^|[[:space:]])echo[[:space:]]+.*>>[[:space:]]*/etc/sudoers
(^|[[:space:]])cat[[:space:]]+.*>[[:space:]]*/etc/passwd
(^|[[:space:]])cat[[:space:]]+.*>[[:space:]]*/etc/shadow

# PAM configuration
(^|[[:space:]])rm[[:space:]]+.*pam\.d
(^|[[:space:]])mv[[:space:]]+.*pam\.d

# sudoers manipulation
(^|[[:space:]])visudo[[:space:]]+
(^|[[:space:]])rm[[:space:]]+.*sudoers
(^|[[:space:]])mv[[:space:]]+.*sudoers
(^|[[:space:]])echo[[:space:]]+.*sudoers

# SSH authorized keys manipulation
(^|[[:space:]])rm[[:space:]]+.*authorized_keys
(^|[[:space:]])cat[[:space:]]+.*>[[:space:]].*authorized_keys
(^|[[:space:]])echo[[:space:]]+.*>[[:space:]].*authorized_keys
(^|[[:space:]])truncate[[:space:]]+.*authorized_keys

# Home directory removal
(^|[[:space:]])rm[[:space:]]+-rf[[:space:]]+/home/
(^|[[:space:]])rm[[:space:]]+-rf[[:space:]]+/Users/
(^|[[:space:]])userdel[[:space:]]+-r

# Login shell manipulation
(^|[[:space:]])chsh[[:space:]]+
(^|[[:space:]])usermod[[:space:]]+-s[[:space:]]+/sbin/nologin
(^|[[:space:]])usermod[[:space:]]+-s[[:space:]]+/bin/false

# Kerberos
(^|[[:space:]])kdestroy[[:space:]]+
(^|[[:space:]])kinit[[:space:]]+.*-k

# LDAP
(^|[[:space:]])ldapdelete[[:space:]]+
(^|[[:space:]])ldapmodify[[:space:]]+.*changetype:[[:space:]]*delete

# macOS user management
(^|[[:space:]])dscl[[:space:]]+.[[:space:]]+delete
(^|[[:space:]])dscl[[:space:]]+.[[:space:]]+delete[[:space:]]+/Users
(^|[[:space:]])dscl[[:space:]]+.[[:space:]]+delete[[:space:]]+/Groups
(^|[[:space:]])sysadminctl[[:space:]]+-deleteUser
