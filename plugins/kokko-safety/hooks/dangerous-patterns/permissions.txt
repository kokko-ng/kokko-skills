# permissions.txt - File permission and ownership operations

# World-writable permissions (security risk)
(^|[[:space:]])chmod[[:space:]]+777
(^|[[:space:]])chmod[[:space:]]+-R[[:space:]]+777
(^|[[:space:]])chmod[[:space:]]+a\+w
(^|[[:space:]])chmod[[:space:]]+-R[[:space:]]+a\+w
(^|[[:space:]])chmod[[:space:]]+o\+w
(^|[[:space:]])chmod[[:space:]]+-R[[:space:]]+o\+w

# Remove all permissions
(^|[[:space:]])chmod[[:space:]]+000
(^|[[:space:]])chmod[[:space:]]+-R[[:space:]]+000

# Recursive chmod on root or system paths (routine project-local chmod -R is fine;
# world-writable recursive modes are covered above)
(^|[[:space:]])chmod[[:space:]]+(-[a-zA-Z]+[[:space:]]+)*(-[a-zA-Z]*R[a-zA-Z]*|--recursive)[[:space:]]+[^[:space:]]+[[:space:]]+/(etc|usr|bin|sbin|lib|var|boot|home|opt|srv)?([[:space:]]|/|$)

# SUID/SGID manipulation (security sensitive)
(^|[[:space:]])chmod[[:space:]]+[ug]\+s
(^|[[:space:]])chmod[[:space:]]+[0-7]?[4-7][0-7][0-7][0-7]
(^|[[:space:]])chmod[[:space:]]+4[0-7][0-7][0-7]
(^|[[:space:]])chmod[[:space:]]+2[0-7][0-7][0-7]
(^|[[:space:]])chmod[[:space:]]+6[0-7][0-7][0-7]

# Sticky bit removal
(^|[[:space:]])chmod[[:space:]]+-t

# World-writable numeric modes (other-write bit set; 0644/0755 etc. are routine)
(^|[[:space:]])chmod[[:space:]]+(-[a-zA-Z]+[[:space:]]+)*[0-7]?[0-7][0-7][2367]([[:space:]]|$)

# Recursive ownership change
(^|[[:space:]])chown[[:space:]]+-R
(^|[[:space:]])chown[[:space:]]+--recursive
(^|[[:space:]])chgrp[[:space:]]+-R
(^|[[:space:]])chgrp[[:space:]]+--recursive

# Change ownership to root
(^|[[:space:]])chown[[:space:]]+root
(^|[[:space:]])chown[[:space:]]+0:

# Change ownership on system directories
(^|[[:space:]])chown[[:space:]]+.*[[:space:]]+/etc
(^|[[:space:]])chown[[:space:]]+.*[[:space:]]+/usr
(^|[[:space:]])chown[[:space:]]+.*[[:space:]]+/var
(^|[[:space:]])chown[[:space:]]+.*[[:space:]]+/bin
(^|[[:space:]])chown[[:space:]]+.*[[:space:]]+/sbin
(^|[[:space:]])chown[[:space:]]+.*[[:space:]]+/lib
(^|[[:space:]])chmod[[:space:]]+.*[[:space:]]+/etc
(^|[[:space:]])chmod[[:space:]]+.*[[:space:]]+/usr
(^|[[:space:]])chmod[[:space:]]+.*[[:space:]]+/var
(^|[[:space:]])chmod[[:space:]]+.*[[:space:]]+/bin
(^|[[:space:]])chmod[[:space:]]+.*[[:space:]]+/sbin
(^|[[:space:]])chmod[[:space:]]+.*[[:space:]]+/lib

# ACL manipulation
(^|[[:space:]])setfacl[[:space:]]+-b
(^|[[:space:]])setfacl[[:space:]]+--remove-all
(^|[[:space:]])setfacl[[:space:]]+-x
(^|[[:space:]])setfacl[[:space:]]+--remove

# Extended attributes
(^|[[:space:]])setattr[[:space:]]+-i
(^|[[:space:]])chattr[[:space:]]+-i
(^|[[:space:]])chattr[[:space:]]+\+i[[:space:]]+/

# macOS specific
(^|[[:space:]])xattr[[:space:]]+-d
(^|[[:space:]])xattr[[:space:]]+-c
(^|[[:space:]])xattr[[:space:]]+--clear

# SELinux
(^|[[:space:]])setenforce[[:space:]]+0
(^|[[:space:]])chcon[[:space:]]+-R
(^|[[:space:]])restorecon[[:space:]]+-R
(^|[[:space:]])semanage[[:space:]]+.*[[:space:]]+--delete
(^|[[:space:]])setsebool[[:space:]]

# AppArmor
(^|[[:space:]])aa-disable
(^|[[:space:]])aa-teardown
(^|[[:space:]])apparmor_parser[[:space:]]+-R

# Capabilities
(^|[[:space:]])setcap[[:space:]]+-r
(^|[[:space:]])setcap[[:space:]]+cap_.*=
