# git.txt - Git destructive operations

# Force push variants
(^|[[:space:]])git[[:space:]]+push[[:space:]]+(.*[[:space:]])?(--force|-f)([[:space:]]|$)
(^|[[:space:]])git[[:space:]]+push[[:space:]]+--force-with-lease

# Hard reset
(^|[[:space:]])git[[:space:]]+reset[[:space:]]+--hard
(^|[[:space:]])git[[:space:]]+reset[[:space:]]+--merge
(^|[[:space:]])git[[:space:]]+reset[[:space:]]+--keep

# Clean with force and directories
(^|[[:space:]])git[[:space:]]+clean[[:space:]]+-[a-zA-Z]*f
(^|[[:space:]])git[[:space:]]+clean[[:space:]]+-[a-zA-Z]*d
(^|[[:space:]])git[[:space:]]+clean[[:space:]]+-[a-zA-Z]*x

# Delete branch (any variant)
(^|[[:space:]])git[[:space:]]+branch[[:space:]]+(-[dD]|--delete)

# Delete remote branch
(^|[[:space:]])git[[:space:]]+push[[:space:]]+(.*[[:space:]])?--delete
(^|[[:space:]])git[[:space:]]+push[[:space:]]+[^[:space:]]+[[:space:]]+:[^[:space:]]

# Interactive rebase (requires interaction)
(^|[[:space:]])git[[:space:]]+rebase[[:space:]]+-i
(^|[[:space:]])git[[:space:]]+rebase[[:space:]]+--interactive

# NOTE: direct pushes to protected branches are owned by
# pre-tool-branch-protection.sh -- do not duplicate them here or the same
# command double-prompts.

# Checkout/restore that discards changes
(^|[[:space:]])git[[:space:]]+checkout[[:space:]]+--[[:space:]]+\.
(^|[[:space:]])git[[:space:]]+checkout[[:space:]]+\.
(^|[[:space:]])git[[:space:]]+restore[[:space:]]+--staged[[:space:]]+\.
(^|[[:space:]])git[[:space:]]+restore[[:space:]]+\.

# Stash destructive operations
(^|[[:space:]])git[[:space:]]+stash[[:space:]]+drop
(^|[[:space:]])git[[:space:]]+stash[[:space:]]+clear
(^|[[:space:]])git[[:space:]]+stash[[:space:]]+pop[[:space:]]+--index

# History rewriting
(^|[[:space:]])git[[:space:]]+filter-branch
(^|[[:space:]])git[[:space:]]+filter-repo
# Destructive rebase forms. The recovery flags --continue/--abort/--skip/
# --quit/--edit-todo are deliberately NOT matched: they are how you get OUT
# of a rebase gone wrong.
(^|[[:space:]])git[[:space:]]+rebase[[:space:]]+(--onto|--root|--autosquash|--exec|-x[[:space:]]|--strategy|-s[[:space:]]|-X|--committer-date-is-author-date|--ignore-date|--ignore-whitespace|--whitespace|--force-rebase|-f([[:space:]]|$))

# Reflog manipulation
(^|[[:space:]])git[[:space:]]+reflog[[:space:]]+expire
(^|[[:space:]])git[[:space:]]+reflog[[:space:]]+delete

# Garbage collection (can remove unreachable objects)
(^|[[:space:]])git[[:space:]]+gc[[:space:]]+--prune
(^|[[:space:]])git[[:space:]]+prune

# Submodule removal
(^|[[:space:]])git[[:space:]]+submodule[[:space:]]+deinit
(^|[[:space:]])git[[:space:]]+rm[[:space:]]+--cached

# Worktree removal
(^|[[:space:]])git[[:space:]]+worktree[[:space:]]+remove
(^|[[:space:]])git[[:space:]]+worktree[[:space:]]+prune

# Tag deletion
(^|[[:space:]])git[[:space:]]+tag[[:space:]]+-d
(^|[[:space:]])git[[:space:]]+tag[[:space:]]+--delete
(^|[[:space:]])git[[:space:]]+push[[:space:]]+.*[[:space:]]+:refs/tags/

# Remote manipulation
(^|[[:space:]])git[[:space:]]+remote[[:space:]]+remove
(^|[[:space:]])git[[:space:]]+remote[[:space:]]+rm

# Config manipulation (security-sensitive)
(^|[[:space:]])git[[:space:]]+config[[:space:]]+--global[[:space:]]+--unset
(^|[[:space:]])git[[:space:]]+config[[:space:]]+--system
