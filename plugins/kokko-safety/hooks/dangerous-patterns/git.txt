# git.txt - Git destructive operations

# Force push variants (any branch, any flag position; git may carry -C <dir>).
# This file is the sole owner of force pushes -- see the ownership note below.
(^|[[:space:]])git[[:space:]]+(-C[[:space:]]+[^[:space:]]+[[:space:]]+)?push[[:space:]]+(.*[[:space:]])?(--force|-f)([[:space:]]|$)
(^|[[:space:]])git[[:space:]]+(-C[[:space:]]+[^[:space:]]+[[:space:]]+)?push[[:space:]]+(.*[[:space:]])?--force-with-lease

# Hard reset
(^|[[:space:]])git[[:space:]]+reset[[:space:]]+--hard
(^|[[:space:]])git[[:space:]]+reset[[:space:]]+--merge
(^|[[:space:]])git[[:space:]]+reset[[:space:]]+--keep

# Clean with force and directories
(^|[[:space:]])git[[:space:]]+clean[[:space:]]+-[a-zA-Z]*f
(^|[[:space:]])git[[:space:]]+clean[[:space:]]+-[a-zA-Z]*d
(^|[[:space:]])git[[:space:]]+clean[[:space:]]+-[a-zA-Z]*x

# Force-delete branch. Plain `git branch -d` / `--delete` refuse on unmerged
# work by themselves and are the cleanup path the janitor skill and the
# kokko-devcontainer guard prescribe -- prompting on them is pure noise, and
# noise gets safety hooks switched off. Only the force forms (-D, -d
# clustered with -f, and --delete combined with --force) drop unmerged
# commits, so only they ask. The case: marker makes the match case-sensitive
# (grep -i cannot otherwise tell -D from -d).
case:(^|[[:space:]])git[[:space:]]+branch[[:space:]]+(-[a-zA-Z]*D|--delete[[:space:]]+--force|--force[[:space:]]+--delete|-[a-zA-Z]*d[a-zA-Z]*f|-[a-zA-Z]*f[a-zA-Z]*d)([[:space:]]|$)

# Delete remote branch
(^|[[:space:]])git[[:space:]]+push[[:space:]]+(.*[[:space:]])?--delete
(^|[[:space:]])git[[:space:]]+push[[:space:]]+[^[:space:]]+[[:space:]]+:[^[:space:]]

# Interactive rebase (requires interaction)
(^|[[:space:]])git[[:space:]]+rebase[[:space:]]+-i
(^|[[:space:]])git[[:space:]]+rebase[[:space:]]+--interactive

# NOTE: division of labor with pre-tool-branch-protection.sh. This file owns
# force push, hard reset, and rebase on ALL branches; branch-protection owns
# commit and plain (non-force) push on protected branches. Do not duplicate
# either side or the same command double-prompts.

# Checkout/restore that discards changes. `restore --staged` WITHOUT
# --worktree only touches the index (it is the standard unstage, and the
# alternative the kokko-devcontainer guard recommends) so it passes; the
# --worktree/-W forms overwrite working-tree files and ask.
(^|[[:space:]])git[[:space:]]+checkout[[:space:]]+--[[:space:]]+\.
(^|[[:space:]])git[[:space:]]+checkout[[:space:]]+\.
(^|[[:space:]])git[[:space:]]+restore[[:space:]]+(.*[[:space:]])?(--worktree|-[a-zA-Z]*W)([[:space:]]|$)
(^|[[:space:]])git[[:space:]]+restore[[:space:]]+\.

# Stash destructive operations
(^|[[:space:]])git[[:space:]]+stash[[:space:]]+drop
(^|[[:space:]])git[[:space:]]+stash[[:space:]]+clear
(^|[[:space:]])git[[:space:]]+stash[[:space:]]+pop[[:space:]]+--index

# History rewriting
(^|[[:space:]])git[[:space:]]+filter-branch
(^|[[:space:]])git[[:space:]]+filter-repo
# Destructive rebase forms. The recovery flags --continue/--abort/--skip/
# --quit/--edit-todo are deliberately NOT matched: they are how you get OUT
# of a rebase gone wrong.
(^|[[:space:]])git[[:space:]]+rebase[[:space:]]+(--onto|--root|--autosquash|--exec|-x[[:space:]]|--strategy|-s[[:space:]]|-X|--committer-date-is-author-date|--ignore-date|--ignore-whitespace|--whitespace|--force-rebase|-f([[:space:]]|$))

# Reflog manipulation
(^|[[:space:]])git[[:space:]]+reflog[[:space:]]+expire
(^|[[:space:]])git[[:space:]]+reflog[[:space:]]+delete

# Garbage collection (can remove unreachable objects)
(^|[[:space:]])git[[:space:]]+gc[[:space:]]+--prune
(^|[[:space:]])git[[:space:]]+prune

# Submodule removal. `git rm --cached` is deliberately NOT matched: it is
# index-only (the file stays on disk) and is the exact alternative the
# kokko-devcontainer guard's `git rm -f` deny message recommends.
(^|[[:space:]])git[[:space:]]+submodule[[:space:]]+deinit

# Forced worktree removal. The plain form refuses on a dirty worktree by
# itself and is the cleanup path the janitor skill prescribes -- only
# --force/-f deletes a worktree along with its uncommitted changes.
(^|[[:space:]])git[[:space:]]+worktree[[:space:]]+remove[[:space:]]+(.*[[:space:]])?(--force|-[a-zA-Z]*f)([[:space:]]|$)
(^|[[:space:]])git[[:space:]]+worktree[[:space:]]+prune

# Tag deletion
(^|[[:space:]])git[[:space:]]+tag[[:space:]]+-d
(^|[[:space:]])git[[:space:]]+tag[[:space:]]+--delete
(^|[[:space:]])git[[:space:]]+push[[:space:]]+.*[[:space:]]+:refs/tags/

# Remote manipulation
(^|[[:space:]])git[[:space:]]+remote[[:space:]]+remove
(^|[[:space:]])git[[:space:]]+remote[[:space:]]+rm

# Config manipulation (security-sensitive)
(^|[[:space:]])git[[:space:]]+config[[:space:]]+--global[[:space:]]+--unset
(^|[[:space:]])git[[:space:]]+config[[:space:]]+--system
